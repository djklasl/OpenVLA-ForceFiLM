flowchart TB

    %% ==================== 第一阶段：启动与续采 ====================
    subgraph P1["🚀 第一阶段：启动与续采"]
        A["程序启动"]
        B["扫描输出目录<br/>判断所有 task 完成度"]
        C{"有未完成的 task?"}

        A --> B --> C
        C -- 否 --> END1["✅ 所有 task 已完成<br/>程序结束"]
        C -- 是 --> D["进入未完成 task<br/>（优先进入到完成一些demo的task）"]
    end

    %% ==================== 第二阶段：任务入口 ====================
    subgraph P2["👀 第二阶段：任务入口与预览"]
        D --> E["任务入口页<br/>┌─────────────┐<br/>│ 播放原始轨迹 │<br/>│    退出      │<br/>└─────────────┘"]

        E -- 点击播放 --> F["查找第一条成功 demo<br/>（失败则 fallback demo_0）"]
        F --> G["🎬 自动播放预览"]
        G --> H["预览完成页<br/>┌─────────────┐<br/>│  数据采集    │<br/>│    重播      │<br/>│    退出      │<br/>└─────────────┘"]

        H -- 重播 --> G
        H -- 退出 --> END2["❌ 程序结束"]
    end

    %% ==================== 第三阶段：采集循环 ====================
    subgraph P3["🔄 第三阶段：单 Demo 采集"]
        H -- 数据采集 --> I["进入待采集 demo"]

        I --> J["🤖 自动回放前段"]
        J --> K{"按 H 切换人工"}

        K --> L["🎮 人工遥操作模式"]
        L --> M["Demo 结束"]
    end

    %% ==================== 第四阶段：结果处理 ====================
    subgraph P4["💾 第四阶段：结果处理"]
        M --> N["写入 HDF5"]
        N --> O{"当前 task 完成?"}

        O -- 否 --> P["下一个 demo"]
        P --> H

        O -- 是 --> Q{"还有其他 task?"}
        Q -- 是 --> R["进入下一个 task"]
        R --> E

        Q -- 否 --> END3["✅ 所有 task 完成"]
    end

    %% ==================== 重启逻辑 ====================
    J -. "R: 重启当前 demo<br/>不保存任何数据" .-> H
    L -. "R: 重启当前 demo<br/>不保存任何数据" .-> H
